% ========== Estilo de todos los entornos (sangría francesa) ==========
% Mismo aspecto que en la plantilla de documento corto (cabecera en
% negrita, cuerpo en cursiva, numeración por sección idéntica), pero
% con sangría francesa: la cabecera abre la primera línea al margen y
% las líneas siguientes quedan sangradas.
\newtheoremstyle{apuntes}
  {}% espacio vertical por encima (por defecto, como en documento corto)
  {}% espacio vertical por debajo (por defecto, como en documento corto)
  {\itshape}% cuerpo en cursiva
  {}% sin sangría extra de la cabecera: la cabecera va al margen
  {\bfseries}% cabecera en negrita
  {.}% puntuación tras la cabecera
  {5pt plus 1pt minus 1pt}% espacio tras la puntuación de la cabecera
  {}% cabecera por defecto: nombre + número + (nota)

% Estilo de las demostraciones: cabecera en cursiva y cuerpo en
% redonda, igual que \begin{proof} de amsthm, pero con la palabra
% «Demostración» metida en la sangría francesa del párrafo (no como
% etiqueta aparte), de modo que abre la primera línea al margen y las
% líneas siguientes quedan sangradas.
\newtheoremstyle{demostracion}
  {}
  {}
  {\normalfont}% cuerpo en redonda
  {}
  {\itshape}% cabecera en cursiva
  {}% SIN puntuación tras la cabecera: el nombre («Dem.:») ya termina
     % en su propia puntuación; dejar aquí un punto produciría «Dem.:.»
  {\labelsep}% espacio tras la cabecera, como el proof original
  {\thmnote{#3}}% la cabecera ES la nota: reproduce que un nombre opcional sustituya a «Demostración»

% Parche de amsthm: aplica la sangría francesa a todos los párrafos
% del entorno (no solo al primero). El mecanismo interno de amsthm
% inserta la cabecera al principio del primer párrafo; los párrafos
% siguientes reactivan la sangría rearmándose.
\makeatletter
% \dth@everypar es un registro toks de amsthm: guardamos su contenido
% original y reescribimos el registro (no se puede \def sobre él).
\newtoks\apuntes@dth@everypar
\apuntes@dth@everypar=\dth@everypar
\dth@everypar{%
  \hangindent\sangriaentorno\hangafter\@ne
  \the\apuntes@dth@everypar
  \everypar\expandafter{\the\dth@everypar}% rearma la sangría para los párrafos siguientes
}
% Fin del parche de la sangría

% Demostración como entorno sin número con símbolo de QED. El nombre
% («Demostración» por defecto) va metido en la sangría francesa: abre
% la primera línea del párrafo y el cuerpo prosigue en ella.
\theoremstyle{demostracion}
\newtheorem*{demostracion@apl}{}

\renewenvironment{proof}[1][\proofname]{%
  \par
  \pushQED{\qed}%
  \def\@demostracion@nota{#1}%
  \expandafter\demostracion@apl\expandafter[\@demostracion@nota]%
}{%
  \popQED
  \enddemostracion@apl
}

% Alias en castellano: \begin{demostracion} y \begin{demostracion}[Nombre]
% son equivalentes a \begin{proof} y \begin{proof}[Nombre]; en los
% apuntes usaremos siempre la forma castellana.
\newenvironment{demostracion}[1][\proofname]{\proof[#1]}{\endproof}

% Resolución de problemas: mismo estilo que la demostración
% (cabecera en cursiva que ES la nota opcional, cuerpo en redonda,
% sangría francesa), pero SIN el cuadrado QED: conforme a mi estilo,
% las resoluciones de problemas no cierran con el cuadrado (solo las
% demostraciones). La cabecera por defecto es «Resolución».
% NO se puede reutilizar demostracion directamente: heredaría el QED
% de proof. Como a las demostraciones, se le pega al problema
% (needspace + penalty de \endparenv, enganchado más abajo con
% \apuntes@ligarend{resolucion@apl}).
\theoremstyle{demostracion}
\newtheorem*{resolucion@apl}{}
\newenvironment{resolucion}[1][Resolución]{%
  \par
  \resolucion@apl[{#1}]%
}{%
  \endresolucion@apl
}

% Las demostraciones van siempre inmediatamente después de su
% resultado: se prohíbe el salto de página justo después de un
% entorno (la demostración correspondiente siempre queda pegada). El
% penalty se inserta dentro de \endtrivlist, antes del glue que este
% emite, porque un penalty posterior no bloquearía el salto en ese
% glue; la redefinición de \@endparenv queda local al entorno.
\let\apuntes@endparenv\@endparenv
\def\@endtheorem{%
  \def\@endparenv{\penalty\@M\apuntes@endparenv}%
  \endtrivlist
  \@endpefalse}
\makeatother

% La definición lleva el término definido en una nota al margen
% IZQUIERDO (no entre paréntesis después de «Def.:»), con el mismo
% aspecto que \notamargen (cursiva, pequeña), alineada hacia el
% texto: \begin{definicion}[Hausdorff] escribe «Hausdorff» en el
% margen izquierdo a la altura de la cabecera. El margen izquierdo
% de la página (2,5 cm) ya reserva sitio para estas notas
% (marginparsep + marginparwidth = 2 cm).
\newtheorem{definicion@apl}{Def.:}[section]
\makeatletter
\newenvironment{definicion}[1][]{%
  % #1 = término que se está definiendo; si se da, va al margen
  % izquierdo. \leavevmode abre el párrafo (con su cabecera «Def.:
  % n.m») para que la nota se ancle a esa misma línea. Para escribir
  % en el margen izquierdo con marginnote hay que invertir el margen
  % (\reversemarginpar) y darle el texto en el argumento OPCIONAL,
  % dejando vacío el obligatorio (así funciona marginnote v1.5); el
  % grupo limita la inversión a esta nota. marginnote ya alinea las
  % notas del margen izquierdo hacia el texto (\raggedleft).
  \definicion@apl
  \if\relax\detokenize{#1}\relax\else
    \leavevmode
    {\reversemarginpar\marginnote[{\footnotesize\itshape #1}]{}}%
  \fi
}{%
  \enddefinicion@apl
}
\makeatother

% range= deben ir los nombres primarios de unicode-math (\triangle

% ========== Red de seguridad para los marcadores PDF ==========
% Al construir los marcadores (bookmarks) del PDF, hyperref convierte
% los títulos de sección a «PDF strings», que no admiten mates ni
% comandos de composición: cualquier título con mates inline
% provocaba avisos «Token not allowed in a PDF string». La regla
% correcta al redactar es \texorpdfstring en cada título con mates
% (véase el README), que además fija la versión del marcador; esta
% red elimina los avisos de los comandos marrones (\mathbb, \emph,
% \textbf, …), mientras que los `$` y `^` que queden siguen dando
% un aviso par de veces por título (inofensivo: hyperref los quita
% y compone el resto). Si algo sobrevive, se omite en el marcador,
% nunca en el título impreso.
\makeatletter
\pdfstringdefDisableCommands{%
    \def\emph#1{#1}%
    \def\textbf#1{#1}%
    \def\textnormal#1{#1}%
    \let\ensuremath\@firstofone
    \let\mathbb\@firstofone
    \let\mathbbm\@firstofone
    \let\mathcal\@firstofone
    \let\mathrm\@firstofone
    \let\operatorname\@firstofone
    \let\vec\@firstofone
    \let\underline\@firstofone
    \let\lVert\relax
    \let\rVert\relax
}
\makeatother
% FIN ========== Red de seguridad para los marcadores PDF ==========

% ========== Entradas robustas de ficheros ==========
% La plantilla busca cada fichero primero en su carpeta (front_matter/ o
% main_matter/) y, si no está ahí (por ejemplo, si subiste los ficheros
% sueltos a la raíz del proyecto de Overleaf), lo busca directamente en
% la raíz: funciona con ambas disposiciones.
\makeatletter
\newcommand{\apuntes@input@dos}[2]{%
  % #1 = carpeta con barra final (p. ej. «front_matter/»), #2 = nombre
  % sin extensión. Si existe <carpeta><nombre>.tex, se incluye desde ahí;
  % si no, se incluye desde la raíz como <nombre>.tex.
  \InputIfFileExists{#1#2.tex}{}{\input{#2}}}
\newcommand{\inputportada}[1]{\apuntes@input@dos{front_matter/}{#1}}
\newcommand{\inputtema}[1]{\apuntes@input@dos{main_matter/}{#1}}
\makeatother
% FIN ========== Entradas robustas de ficheros ==========

% ========== Cada tema empieza en página nueva ==========
% Como un tema es una sección (\section), aquí se consigue que toda
% sección empiece en página nueva, al estilo de los capítulos de un
% libro. Se envuelve \section DESPUÉS de cargar hyperref y cleveref:
% hyperref engancha sus anclajes y marcadores por dentro de \@sect,
% así que el envoltorio se los conserva intactos. Los usos
% estrellados (\section*, como el título del índice) también pasan
% por el envoltorio; el \clearpage extra es inofensivo porque la
% portada ya deja la página rota.
\makeatletter
\let\apuntes@section\section
\renewcommand{\section}{\clearpage\apuntes@section}
\makeatother
% FIN ========== Cada tema empieza en página nueva ==========
